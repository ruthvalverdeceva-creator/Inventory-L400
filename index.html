function guardarRegistro() { const payload = { Operario: operarioSeleccionado, Pasillo: pasilloSeleccionado, // Ej: "Pasillo 1" Ubicacion: document.getElementById('txtUbicacion').value, Codigo: document.getElementById('txtCodigo').value, Descripcion: document.getElementById('txtDescripcion').value, UMA: document.getElementById('txtUMA').value, CamaPqt: parseInt(document.getElementById('txtCama').value) || 0, Niveles: parseInt(document.getElementById('txtNiveles').value) || 0, Saldo: parseInt(document.getElementById('txtSaldo').value) || 0, Total: parseInt(document.getElementById('txtTotal').value) || 0 }; fetch('/api/guardar', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }) .then(async res => { const result = await res.json(); if (!res.ok) { // MUESTRA LA ALERTA DE ERROR DE PASILLO Y DETIENE EL PROCESO alert(result.message); return; } alert("✅ Registro guardado correctamente"); limpiarFormulario(); }) .catch(err => { alert("Error de conexión al guardar el registro."); }); }
